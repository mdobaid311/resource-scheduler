## The code

```tsx
import { useState } from "react";
import { ResourceScheduler, ViewType } from "resource-scheduler";
import "resource-scheduler/dist/resource-scheduler.css";

// Today at the given hour, so the example shows up on whatever week you open it.
const at = (hour: number) => {
  const d = new Date();
  d.setHours(hour, 0, 0, 0);
  return d;
};

export function Board() {
  const [resources, setResources] = useState([
    {
      id: "ann",
      name: "Ann Lee",
      events: [
        {
          id: "e1",
          title: "Standup",
          startDate: at(9),
          endDate: at(10),
          color: "#3b82f6",
        },
      ],
    },
  ]);

  return (
    <div style={{ height: 480 }}>
      <ResourceScheduler
        resources={resources}
        initialView={ViewType.Week}
        onEventCreate={(event, resourceId) =>
          setResources((prev) =>
            prev.map((r) =>
              r.id === resourceId
                ? { ...r, events: [...r.events, { ...event, id: crypto.randomUUID() }] }
                : r
            )
          )
        }
        onEventDrop={(event, fromId, toId, start, end) =>
          setResources((prev) =>
            prev.map((r) => {
              const rest = r.events.filter((e) => e.id !== event.id);
              if (r.id === toId) return { ...r, events: [...rest, { ...event, startDate: start, endDate: end }] };
              return r.id === fromId ? { ...r, events: rest } : r;
            })
          )
        }
        onEventResize={(event, resourceId, start, end) =>
          setResources((prev) =>
            prev.map((r) =>
              r.id === resourceId
                ? { ...r, events: r.events.map((e) => (e.id === event.id ? { ...e, startDate: start, endDate: end } : e)) }
                : r
            )
          )
        }
      />
    </div>
  );
}
```

Two things to notice:

1. **You own the state.** The scheduler calls `onEventCreate` and `onEventDrop` and you update `resources`. It never mutates what you pass in.
2. **Handlers opt features in.** Without `onEventDrop`, a dropped event just snaps back. Without `onEventResize`, there are no resize handles.
