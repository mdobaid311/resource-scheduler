## `ViewType`

A `const` object and a union type of the same name:

```ts
const ViewType = {
  Day: "day",
  Week: "week",
  Month: "month",
  Quarter: "quarter",
  Year: "year",
} as const;
type ViewType = (typeof ViewType)[keyof typeof ViewType];
```

## `Resource` and `SchedulerEvent`

```ts
interface Resource {
  id: string;
  name: string;
  role?: string;
  events?: SchedulerEvent[];             // optional when you pass the flat `events` prop
  businessHours?: BusinessHours | false; // overrides the prop; false = always available
  unavailable?: UnavailableRange[];      // time off, shaded in the grid
  capacity?: number;                     // simultaneous bookings, for utilization only. Default 1
  group?: string;                        // listed under a collapsible header with this name
}

interface BusinessHours {
  daysOfWeek?: number[]; // 0 = Sunday. Default [1, 2, 3, 4, 5]
  startHour?: number;    // day view only. Default 9
  endHour?: number;      // exclusive. Default 17
}

interface UnavailableRange {
  start: Date;
  end: Date;             // exclusive
}

interface SchedulerEvent {
  id: string;
  title: string;
  startDate: Date;
  endDate: Date;      // exclusive
  color?: string;
  description?: string;
  resourceId?: string;    // only in the flat `events` prop: the resource it belongs to
  resourceIds?: string[]; // ...or several
  recurrence?: Recurrence; // makes it repeat
  seriesId?: string;       // set on an occurrence: the id of the event it comes from
}

interface Recurrence {
  freq: "daily" | "weekly" | "monthly" | "yearly";
  interval?: number;     // every n periods. Default 1
  byWeekday?: number[];  // weekly only, 0 = Sunday
  until?: Date;          // last day it may start on, inclusive
  count?: number;        // total occurrences, counting the first
  exceptions?: Date[];   // days to skip
}
```

`Event` is a deprecated alias of `SchedulerEvent`. Handlers receive a `Resource` whose `events` is always an array; the optional `events` above is the input type, exported as `ResourceInput`.

## `SlotSelection`

What `onSlotSelect` receives:

```ts
interface SlotSelection {
  resourceId: string;
  start: Date;
  end: Date;          // exclusive
}
```

## `Placement` and `PlacementRules`

Used by the conflict rules and by `isPlacementAllowed`:

```ts
interface Placement {
  resourceId: string;
  start: Date;
  end: Date;
}

interface PlacementRules {
  eventOverlap?: boolean | ((moving: SchedulerEvent, other: SchedulerEvent) => boolean);
  isValidDrop?: (event: SchedulerEvent, placement: Placement) => boolean;
}
```
